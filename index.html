/**
 * Google Apps Script: Multi-Branch MIS Consolidation & PDF Mailer
 * Author: Shivam Gupta - Senior MIS & Automation Specialist
 */
function runDailyMISConsolidation() {
  const branchFolderId = "1aBcD_BRANCH_SHEETS_FOLDER_ID";
  const masterSheetId = "1xYzW_MASTER_MIS_SHEET_ID";
  const recipientEmails = "management@company.com,ops-head@company.com";
  
  Logger.log("Starting Multi-Branch Sheet Consolidation...");
  const folder = DriveApp.getFolderById(branchFolderId);
  const files = folder.getFilesByType(MimeType.GOOGLE_SHEETS);
  
  let consolidatedData = [];
  let totalRows = 0;
  
  while (files.hasNext()) {
    const file = files.next();
    const branchName = file.getName().replace("MIS_", "");
    const sheet = SpreadsheetApp.open(file).getActiveSheet();
    const lastRow = sheet.getLastRow();
    
    if (lastRow > 1) {
      const data = sheet.getRange(2, 1, lastRow - 1, 8).getValues();
      const enriched = data.map(row => [new Date(), branchName, ...row]);
      consolidatedData.push(...enriched);
      totalRows += data.length;
    }
  }
  
  // Write to Master MIS Sheet
  const masterSheet = SpreadsheetApp.openById(masterSheetId).getSheetByName("Consolidated_Master");
  if (consolidatedData.length > 0) {
    masterSheet.getRange(masterSheet.getLastRow() + 1, 1, consolidatedData.length, consolidatedData[0].length).setValues(consolidatedData);
  }
  
  // Generate Executive PDF & Dispatch via GmailApp
  const pdfBlob = masterSheet.getParent().getAs('application/pdf')
    .setName(`Executive_MIS_Report_${Utilities.formatDate(new Date(), "IST", "yyyy-MM-dd")}.pdf`);
    
  GmailApp.sendEmail(recipientEmails, `Daily Executive MIS Report - ${totalRows} Rows Processed`, 
    "Please find attached the daily consolidated MIS report compiled automatically by Shivam Gupta's Google Apps Script engine.", 
    { attachments: [pdfBlob], name: "Shivam MIS Automation Bot" }
  );
  Logger.log(`Consolidation Complete: ${totalRows} records synced and PDF dispatched.`);
}

'==============================================================================
' Module: modHighSpeedReconciliation
' Author: Shivam Gupta - Senior MIS & Operations Analyst
' Description: Fast in-memory hash reconciliation for 50k+ rows using Scripting.Dictionary
'==============================================================================
Option Explicit

Public Sub RunHighSpeedReconciliation()
    Dim tStart As Double: tStart = Timer
    Dim wsERP As Worksheet, wsPhysical As Worksheet, wsAudit As Worksheet
    Dim dictERP As Object: Set dictERP = CreateObject("Scripting.Dictionary")
    
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    
    Set wsERP = ThisWorkbook.Sheets("ERP_Ledger")
    Set wsPhysical = ThisWorkbook.Sheets("Physical_Audit")
    Set wsAudit = ThisWorkbook.Sheets("Discrepancy_Journal")
    wsAudit.Cells.ClearContents
    
    Dim erpData As Variant: erpData = wsERP.Range("A2:F" & wsERP.Cells(wsERP.Rows.Count, "A").End(xlUp).Row).Value
    Dim i As Long, key As String
    
    ' Build in-memory hash index
    For i = 1 To UBound(erpData, 1)
        key = Trim(CStr(erpData(i, 1))) & "|" & Trim(CStr(erpData(i, 2))) ' SKU | BatchID
        If Not dictERP.Exists(key) Then
            dictERP.Add key, erpData(i, 5) ' Quantity
        End If
    Next i
    
    Dim physData As Variant: physData = wsPhysical.Range("A2:F" & wsPhysical.Cells(wsPhysical.Rows.Count, "A").End(xlUp).Row).Value
    Dim outExceptions() As Variant, excCount As Long: excCount = 0
    ReDim outExceptions(1 To UBound(physData, 1), 1 To 6)
    
    For i = 1 To UBound(physData, 1)
        key = Trim(CStr(physData(i, 1))) & "|" & Trim(CStr(physData(i, 2)))
        If dictERP.Exists(key) Then
            Dim diff As Double: diff = CDbl(physData(i, 5)) - CDbl(dictERP(key))
            If Abs(diff) > 0.001 Then
                excCount = excCount + 1
                outExceptions(excCount, 1) = physData(i, 1)
                outExceptions(excCount, 2) = physData(i, 2)
                outExceptions(excCount, 3) = dictERP(key)
                outExceptions(excCount, 4) = physData(i, 5)
                outExceptions(excCount, 5) = diff
                outExceptions(excCount, 6) = "QTY_MISMATCH"
            End If
        Else
            excCount = excCount + 1
            outExceptions(excCount, 1) = physData(i, 1)
            outExceptions(excCount, 6) = "MISSING_IN_ERP"
        End If
    Next i
    
    If excCount > 0 Then
        wsAudit.Range("A2").Resize(excCount, 6).Value = outExceptions
    End If
    
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    
    MsgBox "Reconciliation Completed in " & Format(Timer - tStart, "0.00") & " seconds!" & vbCrLf & _
           "Exceptions Flagged: " & excCount, vbInformation, "Shivam MIS Engine"
End Sub

// @name         Logistics Portal Dispatch Auto-Macro
// @namespace    https://shivamgupta.portfolio/automation
// @version      2.4.0
// @description  Automated DOM scraping, form autofill, and webhook synchronization
// @author       Shivam Gupta
// @match        https://portal.logistics-ops.internal/*
// @grant        GM_xmlhttpRequest
// ==/UserScript==

(function () {
  'use strict';
  
  const WEBHOOK_URL = 'https://api.mis-internal.com/v1/sync-dispatch';
  
  async function harvestConsignments() {
    console.log('[MIS WebMacro] Scanning DOM for active consignment rows...');
    const rows = document.querySelectorAll('table.dispatch-grid tbody tr');
    const records = [];
    
    rows.forEach(row => {
      const cols = row.querySelectorAll('td');
      if (cols.length >= 6) {
        records.push({
          consignId: cols[0].innerText.trim(),
          clientName: cols[1].innerText.trim(),
          skuCount: parseInt(cols[2].innerText.trim(), 10) || 0,
          status: cols[3].innerText.trim(),
          timestamp: new Date().toISOString()
        });
      }
    });
    
    if (records.length > 0) {
      console.log(`[MIS WebMacro] Found ${records.length} records. Pushing to PostgreSQL webhook...`);
      GM_xmlhttpRequest({
        method: "POST",
        url: WEBHOOK_URL,
        headers: { "Content-Type": "application/json" },
        data: JSON.stringify({ batchId: `BATCH_${Date.now()}`, data: records }),
        onload: (res) => console.log('[MIS WebMacro] Sync Success: HTTP', res.status)
      });
    }
  }

  // Observe Dynamic Single-Page App (SPA) DOM Changes
  const observer = new MutationObserver((mutations) => {
    if (document.querySelector('table.dispatch-grid')) {
      harvestConsignments();
    }
  });
  
  observer.observe(document.body, { childList: true, subtree: true });
})();

"""
Multi-Department Anomaly Detection & AI Executive Briefing Pipeline
Author: Shivam Gupta - Senior MIS & Operations Analyst
"""
import pandas as pd
import numpy as np
import google.generativeai as genai
from datetime import datetime

# Initialize Gemini AI Client
genai.configure(api_key="GEMINI_API_KEY")

def run_mis_etl_pipeline(hr_file: str, sales_file: str):
    print(f"[{datetime.now().strftime('%H:%M:%S')}] Ingesting data via Pandas...")
    df_hr = pd.read_csv(hr_file)
    df_sales = pd.read_csv(sales_file)
    
    # 1. Clean & Compute Attendance & Attrition Z-Scores
    df_hr['att_rate'] = (df_hr['days_present'] / df_hr['total_working_days']) * 100
    hr_mean, hr_std = df_hr['att_rate'].mean(), df_hr['att_rate'].std()
    df_hr['z_score'] = (df_hr['att_rate'] - hr_mean) / hr_std
    anomalies = df_hr[df_hr['z_score'] < -2.0]
    
    # 2. Sales Conversion Velocity
    sales_conversion = (df_sales['deals_closed'].sum() / df_sales['leads_generated'].sum()) * 100
    
    # 3. AI Executive Narrative Generation
    prompt = f"""
    You are an executive MIS analyst. Provide a 3-bullet briefing on today's operational data:
    - Total Headcount: {len(df_hr)}, Average Attendance: {hr_mean:.1f}%
    - HR Outliers: {len(anomalies)} employees flagged with attendance anomalies
    - Sales Conversion: {sales_conversion:.1f}% across {len(df_sales)} sales reps
    Provide proactive root-cause recommendations.
    """
    model = genai.GenerativeModel("gemini-1.5-flash")
    ai_response = model.generate_content(prompt)
    
    print("\n--- EXECUTIVE AI BRIEFING ---")
    print(ai_response.text)
    return {"status": "SUCCESS", "anomalies_detected": len(anomalies), "ai_summary": ai_response.text}

-- =========================================================================
-- PostgreSQL Operational Parameter Persistence Schema & Audit Architecture
-- Author: Shivam Gupta | Senior MIS & Operations Analyst
-- Purpose: Centralize and persist operational thresholds, KPI targets, and
--          audit trails for Power BI, Apps Script, and Python MIS pipelines.
-- =========================================================================

-- 1. Main Parameter Configuration Table
CREATE TABLE IF NOT EXISTS public.mis_config_parameters (
    param_id           SERIAL PRIMARY KEY,
    department         VARCHAR(32) NOT NULL, -- 'HRA', 'SALES', 'INVENTORY'
    param_key          VARCHAR(64) UNIQUE NOT NULL,
    param_value        VARCHAR(128) NOT NULL,
    data_type          VARCHAR(20) DEFAULT 'NUMERIC', -- 'NUMERIC', 'PERCENTAGE', 'INTEGER'
    description        TEXT,
    last_updated_by    VARCHAR(64) DEFAULT 'Shivam Gupta',
    updated_at         TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. Parameter Mutation Audit Trail Table
CREATE TABLE IF NOT EXISTS public.mis_parameter_audit_log (
    audit_id           BIGSERIAL PRIMARY KEY,
    param_key          VARCHAR(64) NOT NULL,
    old_value          VARCHAR(128),
    new_value          VARCHAR(128),
    modified_by        VARCHAR(64),
    modified_at        TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    change_reason      VARCHAR(255) DEFAULT 'User Parameter Update via MIS Console'
);

-- 3. Audit Trigger Function: Automatically logs any parameter change
CREATE OR REPLACE FUNCTION public.fn_audit_parameter_change()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.param_value IS DISTINCT FROM NEW.param_value THEN
        INSERT INTO public.mis_parameter_audit_log (
            param_key, old_value, new_value, modified_by, modified_at, change_reason
        ) VALUES (
            NEW.param_key, OLD.param_value, NEW.param_value, NEW.last_updated_by, CURRENT_TIMESTAMP, 'MIS Console Update'
        );
    END IF;
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_save_parameter_audit
BEFORE UPDATE ON public.mis_config_parameters
FOR EACH ROW EXECUTE FUNCTION public.fn_audit_parameter_change();

    p_key VARCHAR(64),
    p_val VARCHAR(128),
    p_user VARCHAR(64)
)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE public.mis_config_parameters
